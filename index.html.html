function saveJob(name, link){

  const user = auth.currentUser;

  if(!user) return alert("Login first");

  db.ref("saved/"+user.uid).push({

    name:name,

    link:link

  });

  alert("Saved ✅");

}

function loadSavedJobs(){

  const user = auth.currentUser;

  if(!user) return;

  saved.innerHTML = "<h3>⭐ Saved Jobs</h3>";

  db.ref("saved/"+user.uid).once("value", snap=>{

    if(!snap.exists()){

      saved.innerHTML += "<p>No saved jobs yet.</p>";

      return;

    }

    snap.forEach(child=>{

      let job = child.val();

      saved.innerHTML += `

        <div class="card">

          ${job.name}

          <button onclick="window.open('${job.link}')">Open</button>

        </div>

      `;

    });

  });

}
